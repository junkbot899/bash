INSTALLATION LIST - UBUNTU

# vim - cli text editor
sudo apt install vim
# git - version control system
sudo apt install git
# Cmake for compiling projects (open-gl, esp32)
sudo apt install cmake
# Python virtual environment 3.12 for esp32.
sudo apt install python3.12-venv
# tmux - split terminal ui for cli
sudo apt install tmux -y
# productivity apps libre office
# Inkscape, graphic design.
sudo apt update
sudo apt install inkscape
# Blender 3D Graphics object editor.
sudo apt install blender
